### 2.2 创建 MCP Servers

#### 知识库 MCP Server (`app/mcp/servers/knowledge_server.py`)
```python
"""知识库 MCP Server - 提供标准化的知识检索接口"""
import asyncio
from mcp.server import Server
from mcp.server.stdio import stdio_server
from mcp.types import Tool
from app.kb.knowledge_base import KnowledgeBase
from app.core.database import SessionLocal
from app.models.models import KnowledgeDoc

knowledge_server = Server("knowledge-server")

@knowledge_server.list_tools()
async def list_tools() -> list[Tool]:
    return [
        Tool(
            name="retrieve_knowledge",
            description="从课程知识库中检索相关内容",
            inputSchema={
                "type": "object",
                "properties": {
                    "query": {"type": "string", "description": "检索关键词"},
                    "course_id": {"type": "integer", "description": "课程 ID"},
                    "k": {"type": "integer", "description": "返回结果数量", "default": 5}
                },
                "required": ["query", "course_id"]
            }
        ),
        Tool(
            name="list_documents",
            description="列出课程知识库中的所有文档",
            inputSchema={
                "type": "object",
                "properties": {
                    "course_id": {"type": "integer", "description": "课程 ID"}
                },
                "required": ["course_id"]
            }
        )
    ]

@knowledge_server.call_tool()
async def call_tool(name: str, arguments: dict) -> list:
    if name == "retrieve_knowledge":
        kb = KnowledgeBase(arguments["course_id"])
        results = kb.retrieve(arguments["query"], k=arguments.get("k", 5))
        return [{"content": doc.page_content, "score": float(score), "source": doc.metadata.get("source", "")} for doc, score in results]
    elif name == "list_documents":
        db = SessionLocal()
        docs = db.query(KnowledgeDoc).filter(KnowledgeDoc.course_id == arguments["course_id"]).all()
        return [{"id": doc.id, "title": doc.title, "chunk_num": doc.chunk_num} for doc in docs]
    return []

if __name__ == "__main__":
    asyncio.run(knowledge_server.run(stdio_server()))
```

#### 数据库 MCP Server (`app/mcp/servers/database_server.py`)
```python
"""数据库 MCP Server - 提供标准化的数据访问接口"""
import asyncio
from mcp.server import Server
from mcp.server.stdio import stdio_server
from mcp.types import Tool
from app.core.database import SessionLocal
from app.models.models import LearningRecord, Enrollment, Question
from sqlalchemy import func

db_server = Server("database-server")

@db_server.list_tools()
async def list_tools() -> list[Tool]:
    return [
        Tool(
            name="query_student_records",
            description="查询学生学习记录",
            inputSchema={
                "type": "object",
                "properties": {
                    "student_id": {"type": "integer"},
                    "course_id": {"type": "integer"},
                    "limit": {"type": "integer", "default": 20}
                }
            }
        ),
        Tool(
            name="get_course_stats",
            description="获取课程统计数据",
            inputSchema={
                "type": "object",
                "properties": {
                    "course_id": {"type": "integer"}
                },
                "required": ["course_id"]
            }
        )
    ]

@db_server.call_tool()
async def call_tool(name: str, arguments: dict) -> list:
    db = SessionLocal()
    if name == "query_student_records":
        records = db.query(LearningRecord).filter(
            LearningRecord.student_id == arguments["student_id"],
            LearningRecord.course_id == arguments.get("course_id")
        ).limit(arguments.get("limit", 20)).all()
        return [{"question_id": r.question_id, "score": float(r.score) if r.score else None, "is_correct": bool(r.is_correct) if r.is_correct is not None else None} for r in records]
    elif name == "get_course_stats":
        student_count = db.query(func.count(Enrollment.id)).filter(Enrollment.course_id == arguments["course_id"]).scalar()
        question_count = db.query(func.count(Question.id)).filter(Question.course_id == arguments["course_id"]).scalar()
        return [{"student_count": student_count, "question_count": question_count}]
    return []

if __name__ == "__main__":
    asyncio.run(db_server.run(stdio_server()))
```

### 2.3 MCP Client (`app/mcp/client.py`)
```python
"""MCP 客户端 - 管理多个 MCP Server 连接"""
import asyncio
from typing import Any

class MCPClient:
    def __init__(self):
        self.servers: dict[str, dict[str, Any]] = {}
        self._sessions: dict[str, Any] = {}
    
    async def initialize(self):
        await self.register_server("knowledge-server", "python", ["-m", "app.mcp.servers.knowledge_server"])
        await self.register_server("database-server", "python", ["-m", "app.mcp.servers.database_server"])
    
    async def register_server(self, name: str, command: str, args: list[str] | None = None):
        from mcp.client.stdio import stdio_client
        from mcp.client.session import ClientSession
        transport = await stdio_client([command] + (args or []))
        session = ClientSession(transport)
        await session.initialize()
        self._sessions[name] = session
        tools_response = await session.list_tools()
        self.servers[name] = {"tools": [tool.name for tool in tools_response.tools], "session": session}
    
    async def call_tool(self, server_name: str, tool_name: str, arguments: dict) -> Any:
        if server_name not in self._sessions:
            raise ValueError(f"Server {server_name} not registered")
        session = self._sessions[server_name]
        return await session.call_tool(tool_name, arguments)
    
    async def close(self):
        for session in self._sessions.values():
            await session.close()

mcp_client = MCPClient()
```

### 2.4 集成到 TutorAgent (`app/agents/tutor_agent.py`)
```python
"""答疑 Agent - 集成 MCP"""
import asyncio
from app.mcp.client import mcp_client

class TutorAgent:
    def __init__(self, kb, course_id: int):
        self.kb = kb
        self.course_id = course_id
        self.mcp_enabled = True

    async def answer_with_mcp(self, question: str, history: list | None = None):
        if self.mcp_enabled:
            try:
                results = await mcp_client.call_tool(
                    server_name="knowledge-server",
                    tool_name="retrieve_knowledge",
                    arguments={"query": question, "course_id": self.course_id, "k": 5}
                )
                context = "\n\n".join([r["content"] for r in results if "error" not in r])
            except Exception:
                context, _ = query_sources(question, self.kb, k=5)
        else:
            context, _ = query_sources(question, self.kb, k=5)
        
        msgs = _build_prompt(context)
        msgs.extend(_to_role_messages(history or []))
        msgs.append(HumanMessage(content=question))
        model = llm.get_chat_model()
        resp = await model.ainvoke(msgs)
        return resp.content

    def answer(self, question: str, history: list | None = None) -> str:
        return asyncio.run(self.answer_with_mcp(question, history))
```

### 3.2 下拉选择器组件 (`frontend/src/components/form/DropdownSelect.vue`)
```vue
<template>
  <div class="dropdown-select" :class="{ open: isOpen, disabled }">
    <button ref="triggerRef" class="dropdown-trigger" @click="toggle" :disabled="disabled">
      <span v-if="modelValue !== null" class="selected-label">{{ getLabel(modelValue) }}</span>
      <span v-else class="placeholder">{{ placeholder }}</span>
      <ChevronDownIcon class="chevron" :class="{ rotated: isOpen }" />
    </button>
    <Teleport to="body">
      <transition name="dropdown-fade">
        <div v-if="isOpen" ref="popperRef" class="dropdown-popper" :style="popperStyle">
          <div class="dropdown-options">
            <div v-for="option in options" :key="option.value" class="dropdown-option"
                 :class="{ selected: modelValue === option.value }" @click="select(option.value)">
              <span class="option-label">{{ option.label }}</span>
              <CheckIcon v-if="modelValue === option.value" class="check-icon" />
            </div>
          </div>
        </div>
      </transition>
    </Teleport>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted, onUnmounted } from 'vue'
import { onClickOutside } from '@vueuse/core'

interface Option { value: string | number; label: string; disabled?: boolean }
const props = defineProps<{ modelValue: string | number | null; options: Option[]; placeholder?: string; disabled?: boolean }>()
const emit = defineEmits<{ 'update:modelValue': [value: string | number] }>()

const isOpen = ref(false)
const triggerRef = ref<HTMLElement>()
const popperRef = ref<HTMLElement>()
const popperStyle = ref({})

const toggle = () => { if (!props.disabled) { isOpen.value = !isOpen.value; if (isOpen.value) updatePopperPosition() } }
const select = (value: string | number) => { emit('update:modelValue', value); isOpen.value = false }
const getLabel = (value: string | number) => props.options.find(opt => opt.value === value)?.label || ''

const updatePopperPosition = () => {
  if (!triggerRef.value) return
  const rect = triggerRef.value.getBoundingClientRect()
  popperStyle.value = { top: `${rect.bottom + window.scrollY + 4}px`, left: `${rect.left + window.scrollX}px`, minWidth: `${rect.width}px`, zIndex: '1000' }
}

onClickOutside(popperRef, () => { isOpen.value = false })
onMounted(() => { window.addEventListener('scroll', updatePopperPosition, true) })
onUnmounted(() => { window.removeEventListener('scroll', updatePopperPosition, true) })
</script>
```
